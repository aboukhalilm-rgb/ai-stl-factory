app = Flask(__name__)
app.secret_key = get_setting("APP_SECRET_KEY", "change_this_secret_key")


def is_auth_enabled() -> bool:
    return bool(get_setting("APP_PASSWORD", ""))


def require_login() -> bool:
    if not is_auth_enabled():
        return True
    return session.get("authenticated") is True


@app.before_request
def ensure_setup() -> None:
    if request.path in ["/setup", "/login", "/static", "/favicon.ico"]:
        return
    if not is_auth_enabled():
        return redirect(url_for("setup_page"))


@app.route("/setup", methods=["GET", "POST"])
def setup_page():
    if request.method == "POST":
        password = request.form.get("password", "").strip()
        if not password:
            return render_template_string("<h3>Password required</h3><a href='/setup'>Back</a>")
        write_env({"APP_PASSWORD": password, "APP_SECRET_KEY": get_setting("APP_SECRET_KEY", "_change_me_")})
        session["authenticated"] = True
        return redirect(url_for("index"))

    return """
    <html><body>
      <h2>Initial setup</h2>
      <form method="post">
        <label>Password: <input type="password" name="password"></label>
        <button type="submit">Save</button>
      </form>
    </body></html>
    """


@app.route("/login", methods=["POST", "GET"])
def login_page():
    if request.method == "POST":
        password = request.form.get("password", "")
        if password == get_setting("APP_PASSWORD", ""):
            session["authenticated"] = True
            return redirect(url_for("index"))
        return "Invalid password"
    return """
    <html><body>
      <h2>Login</h2>
      <form method="post">
        <input type="password" name="password">
        <button type="submit">Login</button>
      </form>
    </body></html>
    """


@app.route("/logout")
def logout():
    session.pop("authenticated", None)
    return redirect(url_for("login_page"))

@app.route("/api/generate", methods=["POST"])
def generate_manual():
    if not require_login():
        return jsonify({"error": "Authentication required"}), 401

    category = request.form.get("category", "")
    count = max(1, min(int(request.form.get("count", 1)), 10))
    size_mm = request.form.get("size_mm")
    custom_parameters = request.form.get("custom_parameters", "")
    try:
        custom_params = json.loads(custom_parameters) if custom_parameters else {}
    except Exception:
        custom_params = {"raw": custom_parameters}
    if size_mm:
        custom_params["size_mm"] = float(size_mm)

    generator = STLGenerator(GroqKeyRotator())
    request_obj = GenerationRequest(category=category, count=count, custom_parameters=custom_params)
    try:
        result = generator.generate_model(request_obj)
        return jsonify({"status": "ok", "result": result})
    except Exception as exc:
        return jsonify({"status": "failed", "error": str(exc)}), 500


@app.route("/api/settings", methods=["POST"])
def settings_update():
    if not require_login():
        return jsonify({"error": "Authentication required"}), 401

    payload = {}
    for key in [
        "APP_PASSWORD",
        "GROQ_API_KEYS",
        "GROQ_DEFAULT_MODEL",
        "TELEGRAM_BOT_TOKEN",
        "TELEGRAM_CHAT_ID",
        "GOOGLE_DRIVE_FOLDER_ID",
        "APP_SECRET_KEY",
        "APP_PORT",
        "APP_NAME",
    ]:
        if key in request.form:
            value = request.form.get(key, "").strip()
            if value:
                payload[key] = value
    if not payload:
        return jsonify({"status": "noop"})
    write_env(payload)
    app.secret_key = get_setting("APP_SECRET_KEY", app.secret_key)
    return redirect(url_for("index"))


@app.route("/api/scheduler/start")
def scheduler_start():
    if not require_login():
        return jsonify({"error": "Authentication required"}), 401

    scheduler = STLFactoryScheduler()
    scheduler.start()
    return jsonify({"status": "started"})


@app.route("/api/scheduler/stop")
def scheduler_stop():
    if not require_login():
        return jsonify({"error": "Authentication required"}), 401

    return jsonify({"status": "stop requested"})
